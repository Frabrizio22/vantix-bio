function renderProducts(filter='all'){
const grid=document.getElementById('productsGrid');
const filtered=filter==='all'?products:products.filter(p=>p.category.includes(filter));
const now=new Date();
const saleStart=new Date('2026-07-01T00:00:00-07:00');
const saleEnd=new Date('2026-07-05T23:59:59-07:00');
const isSaleActive=(now>=saleStart&&now<=saleEnd);
grid.innerHTML=filtered.map(p=>{
const oos=!p.inStock;
const salePrice=isSaleActive?(p.price*0.8).toFixed(0):null;
return `
<div class="product-card" onclick="window.location.href='${p.url}'">
  ${p.image?`<img src="${p.image}" alt="${p.name}" class="product-image" loading="lazy">`:`<div class="product-image"></div>`}
  ${oos?'<div class="oos-label">Out of Stock</div>':''}
  <div class="product-category">${p.categoryLabel||''}</div>
  <div class="product-name">${p.name}</div>
  <div class="product-desc">${p.desc||''}</div>
  <div class="product-meta">${p.meta||''}</div>
  ${isSaleActive&&!oos?`
    <div class="sale-price">
      <div class="product-price" style="font-size:28px">$${salePrice}</div>
      <div class="original-price">$${p.price}</div>
      <div class="sale-badge">20% OFF</div>
    </div>
  `:`<div class="product-price">$${p.price}</div>`}
  <div class="product-cta">
    ${oos?`<button class="btn btn-dark btn-waitlist" onclick="event.stopPropagation();openWaitlist('${p.sku}','${p.name.replace(/'/g,"\\'")}')">Join Waitlist</button>`:`<button class="btn btn-dark" onclick="event.stopPropagation();addToCart('${p.sku}','${p.name.replace(/'/g,"\\'")}',${salePrice||p.price})">Add to Cart</button>`}
    ${p.coa?`<a href="${p.coa}" class="coa-link" onclick="event.stopPropagation()" target="_blank">COA ↗</a>`:''}
  </div>
</div>
`}).join('');
}
